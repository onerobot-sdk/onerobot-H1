MoveIt2 运动控制 ​

提供标准的 MoveIt2 动作接口，用于双臂的轨迹规划与执行。支持关节空间目标、位姿目标、笛卡尔路径，以及预置命名姿态。

接口总览 ​

接口消息类型说明
/move_groupmoveit_msgs/action/MoveGroupMoveGroup 动作服务器（规划 + 执行）
/execute_trajectorymoveit_msgs/action/ExecuteTrajectory执行预先规划好的轨迹

规划组参考 ​

规划组自由度末端 link说明
left_arm7left_link7左臂
right_arm7right_link7右臂
both_arms14left_link7 + right_link7双臂联合

关节命名约定 ​

左臂关节名为 joint1-l ~ joint7-l，右臂为 joint1-r ~ joint7-r。下发轨迹的 joint_names 必须使用这套带后缀的名称。

预置命名姿态 ​

可直接用名字作为目标，无需手动指定关节角：

名称适用组含义
homeleft_arm / right_arm单臂归零位
readyleft_arm / right_arm单臂常用准备位
dual_homeboth_arms双臂同时归零
dual_readyboth_arms双臂同时进入准备位

C++ 调用方式：

cpp
move_group_->setNamedTarget("ready");
move_group_->move();

C++ 示例 ​

高级接口（推荐） ​

cpp
#include <moveit/move_group_interface/move_group_interface.hpp>

class MoveIt2Controller : public rclcpp::Node {
public:
    MoveIt2Controller() : Node("moveit2_controller") {
        move_group_ = std::make_shared<moveit::planning_interface::MoveGroupInterface>(
            shared_from_this(), "left_arm");
    }

    void executePoseGoal(const geometry_msgs::msg::Pose& target_pose) {
        move_group_->setPoseTarget(target_pose);

        moveit::planning_interface::MoveGroupInterface::Plan plan;
        auto result = move_group_->plan(plan);

        if (result == moveit::core::MoveItErrorCode::SUCCESS) {
            move_group_->execute(plan);
        }
    }

    void executeJointGoal(const std::vector<double>& joints) {
        // joints 顺序与 left_arm 的关节顺序一致：joint1-l ~ joint7-l
        move_group_->setJointValueTarget(joints);

        moveit::planning_interface::MoveGroupInterface::Plan plan;
        if (move_group_->plan(plan) == moveit::core::MoveItErrorCode::SUCCESS) {
            move_group_->execute(plan);
        }
    }

    void executeNamedTarget(const std::string& name) {
        // 例如 "home" / "ready"
        move_group_->setNamedTarget(name);
        move_group_->move();
    }

    void executeCartesianPath(const std::vector<geometry_msgs::msg::Pose>& waypoints) {
        moveit_msgs::msg::RobotTrajectory trajectory;
        double fraction = move_group_->computeCartesianPath(waypoints, 0.01, 0.0, trajectory);

        if (fraction > 0.9) {
            move_group_->execute(trajectory);
        }
    }

private:
    moveit::planning_interface::MoveGroupInterface::SharedPtr move_group_;
};

双臂联动 ​

both_arms 支持位姿目标、关节角目标、命名姿态：

cpp
auto dual = std::make_shared<moveit::planning_interface::MoveGroupInterface>(
    shared_from_this(), "both_arms");

// 1. 位姿目标：左右 tip 各设置一次
dual->setPoseTarget(left_target_pose,  "left_link7");
dual->setPoseTarget(right_target_pose, "right_link7");
dual->move();

// 2. 关节角目标，顺序：joint1-l ~ joint7-l, joint1-r ~ joint7-r
std::vector<double> joints(14, 0.0);
dual->setJointValueTarget(joints);
dual->move();

// 3. 预置命名姿态
dual->setNamedTarget("dual_ready");
dual->move();

底层 Action Client ​

cpp
#include <moveit_msgs/action/move_group.hpp>
#include <rclcpp_action/rclcpp_action.hpp>

class MoveIt2ActionClient : public rclcpp::Node {
public:
    MoveIt2ActionClient() : Node("moveit2_action_client") {
        action_client_ = rclcpp_action::create_client<moveit_msgs::action::MoveGroup>(
            shared_from_this(), "/move_group");
    }

    void sendGoal(const std::string& group_name, const geometry_msgs::msg::Pose& target_pose) {
        if (!action_client_->wait_for_action_server(std::chrono::seconds(10))) {
            RCLCPP_ERROR(get_logger(), "Action server not available");
            return;
        }

        moveit_msgs::action::MoveGroup::Goal goal;
        goal.request.group_name = group_name;
        goal.request.allowed_planning_time = 5.0;
        // 设置 pose 目标...

        auto send_options = rclcpp_action::Client<moveit_msgs::action::MoveGroup>::SendGoalOptions();
        send_options.feedback_callback = [this](auto, auto feedback) {
            RCLCPP_INFO(get_logger(), "State: %s", feedback->feedback.state.c_str());
        };
        send_options.result_callback = [this](auto) {
            RCLCPP_INFO(get_logger(), "Goal completed");
        };

        action_client_->async_send_goal(goal, send_options);
    }

private:
    rclcpp_action::Client<moveit_msgs::action::MoveGroup>::SharedPtr action_client_;
};

Python 示例 ​

Python 端可直接通过 rclpy 的 Action Client 与 /move_group 通信。

python
import rclpy
from rclpy.action import ActionClient
from rclpy.node import Node
from geometry_msgs.msg import Pose
from moveit_msgs.action import MoveGroup
from moveit_msgs.msg import (
    Constraints, JointConstraint, MotionPlanRequest, PlanningOptions,
)


class MoveIt2Client(Node):
    def __init__(self):
        super().__init__("moveit2_client")
        self._client = ActionClient(self, MoveGroup, "/move_group")

    def send_joint_goal(self, group_name: str, joint_names: list, positions: list):
        """关节空间目标。"""
        goal = MoveGroup.Goal()
        req = MotionPlanRequest()
        req.group_name = group_name
        req.allowed_planning_time = 5.0
        req.max_velocity_scaling_factor = 0.1
        req.max_acceleration_scaling_factor = 0.1

        constraints = Constraints()
        for name, pos in zip(joint_names, positions):
            jc = JointConstraint()
            jc.joint_name = name
            jc.position = pos
            jc.tolerance_above = 0.01
            jc.tolerance_below = 0.01
            jc.weight = 1.0
            constraints.joint_constraints.append(jc)
        req.goal_constraints.append(constraints)

        goal.request = req
        goal.planning_options = PlanningOptions()
        goal.planning_options.plan_only = False

        self._client.wait_for_server()
        future = self._client.send_goal_async(goal, feedback_callback=self._on_feedback)
        future.add_done_callback(self._on_goal_response)

    def _on_feedback(self, feedback):
        self.get_logger().info(f"state: {feedback.feedback.state}")

    def _on_goal_response(self, future):
        handle = future.result()
        if not handle.accepted:
            return
        handle.get_result_async().add_done_callback(self._on_result)

    def _on_result(self, future):
        result = future.result().result
        self.get_logger().info(f"error_code: {result.error_code.val}")


def main():
    rclpy.init()
    node = MoveIt2Client()
    node.send_joint_goal(
        "left_arm",
        ["joint1-l", "joint2-l", "joint3-l", "joint4-l", "joint5-l", "joint6-l", "joint7-l"],
        [0.0, 0.5, 0.0, -1.0, 0.0, 0.5, 0.0],
    )
    rclpy.spin(node)
    rclpy.shutdown()


if __name__ == "__main__":
    main()

速度缩放与限位 ​

关节级最大速度上限约 3.14 rad/s（180°/s），最大加速度上限约 5.0 rad/s²。max_velocity_scaling_factor 和 max_acceleration_scaling_factor 是在此基础上的整体缩放。

场景max_velocity_scaling_factormax_acceleration_scaling_factor
安全测试0.05 - 0.10.05 - 0.1
正常操作0.1 - 0.30.1 - 0.3
高速模式0.5 - 1.00.5 - 1.0

⚠️ 注意：首次使用时建议从 0.1 开始逐步调整；轨迹结束位置容差为 0.10 rad，过激进的速度会因末端抖动触发 GOAL_TOLERANCE_VIOLATED。

规划器说明 ​

默认规划器为 RRTConnect（快速、稳定，适合大多数臂部运动）。每个规划组也允许在 goal 的 planner_id 中切换为 RRTstar 或 PRM。
